Run this single script in your terminal from the project root. It checks everything that must be true before any Bob session starts.

python3 - <<'PYEOF'
import yaml, re, os, stat, sys

VALID_CLI = {'read','edit','command','browser'}
issues = []

# 1. Config directory exists
cfg = os.path.expanduser('~/.bob')
if not os.path.isdir(cfg):
    issues.append('~/.bob directory not found')

# 2. custom_modes.yaml exists at global path
cm = os.path.join(cfg, 'custom_modes.yaml')
if not os.path.isfile(cm):
    issues.append('~/.bob/custom_modes.yaml not found — run scripts/install.sh first')
else:
    with open(cm) as f:
        data = yaml.safe_load(f)
    modes = {m['slug']: m for m in data.get('customModes', [])}

    for slug in ('knowledge-manager', 'repo-analyzer'):
        if slug not in modes:
            issues.append(f'Mode missing: {slug}')
        else:
            m = modes[slug]
            flat = [g[0] if isinstance(g,list) else g for g in m.get('groups',[])]
            if 'command' not in flat:
                issues.append(f'{slug}: missing "command" group — shell will be silent')
            if 'read' not in flat:
                issues.append(f'{slug}: missing "read" group')
            if 'edit' not in flat:
                issues.append(f'{slug}: missing "edit" group')
            for g in m.get('groups',[]):
                if isinstance(g, list) and len(g) > 1:
                    rx = g[1].get('fileRegex','')
                    try: re.compile(rx)
                    except re.error as e:
                        issues.append(f'{slug}: bad fileRegex {rx!r}: {e}')
            # No stale save_memory calls
            ci = m.get('customInstructions','') or ''
            for line in ci.splitlines():
                if 'save_memory' in line and 'no save_memory' not in line:
                    issues.append(f'{slug}: stale save_memory instruction: {line.strip()!r}')

# 3. settings.json exists (context auto-load)
sj = os.path.join(cfg, 'settings.json')
if not os.path.isfile(sj):
    issues.append('~/.bob/settings.json not found (context auto-load won\'t work)')

# 4. install.sh is executable
if not (os.stat('scripts/install.sh').st_mode & stat.S_IXUSR):
    issues.append('scripts/install.sh is not executable')

if issues:
    print(f'FAIL — {len(issues)} pre-condition(s) not met:')
    for i in issues: print(f'  - {i}')
    sys.exit(1)
else:
    print('PASS — all pre-conditions met')

Phase 7 Negative tests — known failure modes confirmed absent Automated

These tests confirm that the bugs found in the audit are confirmed absent from the installed file.

python3 - <<'PYEOF'
import yaml, re, sys, os

results = []

with open(os.path.expanduser('~/.bob/custom_modes.yaml')) as f:
    data = yaml.safe_load(f)
our = {m['slug']: m for m in data['customModes']
       if m['slug'] in ('knowledge-manager','repo-analyzer')}

for slug, m in our.items():
    flat = [g[0] if isinstance(g,list) else g for g in m.get('groups',[])]
    ci = m.get('customInstructions','') or ''

    # Bug 1: 'execute' instead of 'command' in CLI file
    if 'execute' in flat:
        results.append(('FAIL', slug, '"execute" group found — regression from IDE port'))
    else:
        results.append(('PASS', slug, '"execute" absent (correct CLI group is "command")'))

    # Bug 2: save_memory as an actionable instruction
    for line in ci.splitlines():
        if 'save_memory' in line and 'no save_memory' not in line and 'save_memory tool in Bob IDE' not in line:
            results.append(('FAIL', slug, f'save_memory instruction: {line.strip()!r}'))
    else:
        results.append(('PASS', slug, 'no stale save_memory instruction'))

    # Bug 3: fileRegex must compile
    for g in m.get('groups',[]):
        if isinstance(g, list) and len(g) > 1:
            rx = g[1].get('fileRegex','')
            try:
                re.compile(rx)
                results.append(('PASS', slug, f'fileRegex {rx!r} compiles'))
            except re.error as e:
                results.append(('FAIL', slug, f'fileRegex {rx!r} broken: {e}'))

for status, slug, msg in results:
    print(f'{status}: [{slug}] {msg}')

failed = [r for r in results if r[0]=='FAIL']
if failed:
    print(f'\n{len(failed)} regression(s) found')
    sys.exit(1)
else:
    print(f'\nAll {len(results)} negative tests passed')

Known Limitations — Bob Shell 1.0.6

fileRegex enforcement (T2-C) Whether Bob 1.0.6 actually enforces the fileRegex restriction at runtime is unverified. The schema accepts it; enforcement depends on the Bob version. Treat T2-C as informational — document the result but do not gate on it.
--no-interactive flag Phase 1 commands use --no-interactive. If your Bob build does not support this flag, run the prompts in an interactive session manually. The acceptance criteria are identical.
browser group not tested The browser group is declared in both modes. There is no automated test for it here because web access requires a real URL and live network. If web-search is a workflow requirement, add a T2-E manually: prompt Bob to fetch a URL and confirm it returns content.
Bob IDE vs Bob Shell This playbook covers Bob Shell CLI only. The Bob IDE path uses execute/skill groups in .bob/custom_modes.yaml — those are verified separately by the adversarial audit script and need no CLI validation.
